{{- /* SPDX-FileCopyrightText: 2026 Christopher Goes */ -}} {{- /* SPDX-License-Identifier: MIT */ -}} {{- /* The install instructions, one tab for each system or way of installing, shared by the Quickstart and Installation pages so that the two never differ. With no argument, they are the download tabs (Windows, macOS, Linux): a download button, then what to click, with no command line. With "command-line", they are the lines to paste into a terminal (Windows, macOS, Linux (any), Linux (apt), Linux (RPM), Go). Hextra's tabs draw each set, synced: a tab picked on one page is picked on the other, and the two sets, whose tab names differ, are picked apart. js/install-platform.js picks the reader's system in each set they have not picked before. Each tab's text is Markdown in assets/install/.md, where @VERSION@ becomes the latest release's version (the version shortcode's), so that every download names a file that exists. @DOWNLOAD-@, a paragraph of its own, becomes the download buttons for that kind of file, drawn here (install-download below; assets/css/custom.css styles them). @APP-MENU@ in the packages' tabs becomes the sentence in assets/install/app-menu.md. In the Linux (RPM) command line, @KEYURL@ becomes the signing key's address on this site and @FINGERPRINT@ its fingerprint. The build fails if a placeholder is missing from its file or left in a tab. {{< install-tabs >}} {{< install-tabs "command-line" >}} */ -}} {{- $set := .Get 0 | default "download" -}} {{- $sets := dict "download" (slice (dict "name" "Windows" "file" "windows-installer" "downloads" (slice "INSTALLER")) (dict "name" "macOS" "file" "macos-app" "downloads" (slice "APP")) (dict "name" "Linux" "file" "linux-packages" "downloads" (slice "DEB" "RPM") "appMenu" true) ) "command-line" (slice (dict "name" "Windows" "file" "windows") (dict "name" "macOS" "file" "macos") (dict "name" "Linux (any)" "file" "linux") (dict "name" "Linux (apt)" "file" "linux-apt" "appMenu" true) (dict "name" "Linux (RPM)" "file" "linux-rpm" "appMenu" true "signingKey" true) (dict "name" "Go" "file" "go") ) -}} {{- $items := index $sets $set -}} {{- if not $items -}} {{- errorf "%s: install-tabs: unknown argument %q (give none, or \"command-line\")" .Position $set -}} {{- end -}} {{- $version := partial "latest-version.html" . -}} {{- $repo := site.Params.repo -}} {{- if not $repo -}} {{- errorf "%s: install-tabs: hugo.yaml has no params.repo" .Position -}} {{- end -}} {{- /* The download buttons for each kind of file, each with its words and its release file. The packages have one file for each kind of computer, so two buttons. The Windows installer is one file for both kinds of computer, so one button, with no kind of computer. The Mac app is one file too, but its two buttons name the reader's kind of Mac, and js/install-platform.js shows only the one that fits when it can tell. The release files are named in .goreleaser.yaml, windows.yml and macos.yml. */ -}} {{- $installer := printf "wopr_%s_windows_setup.exe" $version -}} {{- $app := printf "wopr_%s_macos.dmg" $version -}} {{- $downloads := dict "INSTALLER" (slice (dict "label" "Download Windows installer" "file" $installer)) "APP" (slice (dict "arch" "arm64" "label" "Download macOS app (Apple Silicon)" "file" $app) (dict "arch" "x86-64" "label" "Download macOS app (Intel)" "file" $app)) "DEB" (slice (dict "arch" "x86-64" "label" "Download .deb (Intel/AMD)" "file" (printf "wopr_%s-1_amd64.deb" $version)) (dict "arch" "arm64" "label" "Download .deb (ARM)" "file" (printf "wopr_%s-1_arm64.deb" $version))) "RPM" (slice (dict "arch" "x86-64" "label" "Download .rpm (Intel/AMD)" "file" (printf "wopr-%s-1.x86_64.rpm" $version)) (dict "arch" "arm64" "label" "Download .rpm (ARM)" "file" (printf "wopr-%s-1.aarch64.rpm" $version))) -}} {{- $appMenu := partial "inline/install-tab-file.html" (dict "file" "app-menu" "position" .Position) | strings.TrimSpace -}} {{- if not $appMenu -}} {{- errorf "%s: install-tabs: assets/install/app-menu.md is empty" .Position -}} {{- end -}} {{- $appMenu = printf " %s" $appMenu -}} {{- /* The signing key's address and fingerprint, for the Linux (RPM) command line. The key is published (at /wopr-signing-key.asc) because this page asks for its address. */ -}} {{- $keyURL := "" -}} {{- with resources.Get "wopr-signing-key.asc" -}} {{- $keyURL = .Permalink -}} {{- else -}} {{- errorf "%s: install-tabs: packaging/wopr-signing-key.asc is missing" .Position -}} {{- end -}} {{- $fingerprint := partial "signing-fingerprint.html" . -}} {{- $tabs := slice -}} {{- $buttons := slice -}} {{- range $i, $item := $items -}} {{- $file := $item.file -}} {{- $text := partial "inline/install-tab-file.html" (dict "file" $file "position" $.Position) -}} {{- if $item.appMenu -}} {{- if not (strings.Contains $text "@APP-MENU@") -}} {{- errorf "%s: install-tabs: assets/install/%s.md has no @APP-MENU@" $.Position $file -}} {{- end -}} {{- $text = replace $text "@APP-MENU@" $appMenu -}} {{- end -}} {{- if $item.signingKey -}} {{- range slice "@KEYURL@" "@FINGERPRINT@" -}} {{- if not (strings.Contains $text .) -}} {{- errorf "%s: install-tabs: assets/install/%s.md has no %s" $.Position $file . -}} {{- end -}} {{- end -}} {{- $text = replace $text "@KEYURL@" $keyURL -}} {{- $text = replace $text "@FINGERPRINT@" $fingerprint -}} {{- end -}} {{- $text = replace $text "@VERSION@" $version -}} {{- $rest := $text -}} {{- range $item.downloads -}} {{- $placeholder := printf "@DOWNLOAD-%s@" . -}} {{- if or (ne (strings.Count $placeholder $text) 1) (not (strings.Contains (string ($text | markdownify)) (printf "

%s

" $placeholder))) -}} {{- errorf "%s: install-tabs: assets/install/%s.md must have %s once, as a paragraph of its own" $.Position $file $placeholder -}} {{- end -}} {{- $rest = replace $rest $placeholder "" -}} {{- $buttons = $buttons | append . -}} {{- end -}} {{- with findRE `@[A-Z][A-Z-]*@` $rest 1 -}} {{- errorf "%s: install-tabs: assets/install/%s.md has %s, which nothing fills" $.Position $file (index . 0) -}} {{- end -}} {{- $tabs = $tabs | append (dict "id" $i "name" $item.name "icon" "" "content" $text "selected" false) -}} {{- end -}} {{- /* Hextra's tabs render each tab's Markdown, and raw HTML in it would be dropped (hugo.yaml), so the buttons go into the HTML that comes out, each in place of its placeholder's paragraph. */ -}} {{- $html := string (partial "shortcodes/tabs" (dict "tabs" $tabs "enableSync" true "id" .Ordinal)) -}} {{- range $buttons -}} {{- $paragraph := printf "

@DOWNLOAD-%s@

" . -}} {{- if not (strings.Contains $html $paragraph) -}} {{- errorf "%s: install-tabs: @DOWNLOAD-%s@ is not a paragraph of its own" $.Position . -}} {{- end -}} {{- $button := string (partial "inline/install-download.html" (dict "download" (index $downloads .) "repo" $repo "version" $version)) -}} {{- $html = replace $html $paragraph $button -}} {{- end -}}
{{- $html | safeHTML -}}
